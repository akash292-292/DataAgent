class ConversationMemory:
    def __init__(self, max_sessions=100, session_timeout=3600):  # 1 hour timeout
        self.sessions = {}
        self.max_sessions = max_sessions
        self.session_timeout = session_timeout
    
    def _cleanup_old_sessions(self):
        """Remove expired sessions"""
        current_time = time.time()
        expired_sessions = []
        
        for session_id, session_data in self.sessions.items():
            if current_time - session_data['last_activity'] > self.session_timeout:
                expired_sessions.append(session_id)
        
        for session_id in expired_sessions:
            del self.sessions[session_id]
    
    def get_session(self, session_id: str) -> Dict:
        """Get or create session with cleanup"""
        # Periodic cleanup
        if len(self.sessions) > self.max_sessions * 0.8:
            self._cleanup_old_sessions()
        
        if session_id not in self.sessions:
            self.sessions[session_id] = {
                'conversation_history': [],
                'last_activity': time.time(),
                'created_at': time.time()
            }
        else:
            self.sessions[session_id]['last_activity'] = time.time()
        
        return self.sessions[session_id]
    
    def add_message(self, session_id: str, role: str, content: str):
        """Add message to conversation history"""
        session = self.get_session(session_id)
        message = {
            'role': role,
            'content': content,
            'timestamp': time.time()
        }
        session['conversation_history'].append(message)
        
        # Keep only last 10 messages to prevent memory bloat
        if len(session['conversation_history']) > 30:
            session['conversation_history'] = session['conversation_history'][-10:]
    
    def get_conversation_context(self, session_id: str, max_messages: int = 5) -> str:
        """Get recent conversation context for LLM"""
        session = self.get_session(session_id)
        history = session['conversation_history'][-max_messages:]
        
        if not history:
            return ""
        
        context_parts = ["Previous conversation:"]
        for msg in history:
            role = "User" if msg['role'] == 'user' else "Assistant"
            context_parts.append(f"{role}: {msg['content']}")
        
        return "\n".join(context_parts)

    def __init__(self):

    def _analyze_question_intent(self, question: str) -> Dict[str, str]:
        """Intelligently analyze what kind of response format the user needs"""
        question_lower = question.lower()
        
        # Questions that should use MIXED format (paragraphs + bullet points)
        mixed_format_keywords = [
            'approach', 'framework', 'methodology', 'strategy', 'plan',
            'execution', 'implementation', 'process', 'procedure'
        ]
        
        # Questions that should use primarily PARAGRAPHS
        paragraph_keywords = [
            'explain', 'describe', 'discuss', 'analyze', 'elaborate',
            'what is', 'how does', 'why is', 'what are the implications',
            'background', 'overview', 'concept', 'theory'
        ]
        
        # Questions that should use primarily BULLET POINTS
        bullet_keywords = [
            'list', 'points', 'bullet', 'items', 'steps', 'tasks',
            'key features', 'benefits', 'advantages', 'components',
            'elements', 'factors', 'criteria', 'what are the'
        ]
        
        # Determine primary format
        if any(word in question_lower for word in mixed_format_keywords):
            return "mixed"
        elif any(word in question_lower for word in bullet_keywords):
            return "bullet"
        elif any(word in question_lower for word in paragraph_keywords):
            return "paragraph"
        else:
            return "mixed"  # Default to mixed for balanced responses

    def _clean_response_formatting(self, text: str) -> str:
        """Remove all markdown formatting and ensure clean text"""
        import re
        
        # Remove **bold** and *italic* markdown
        text = re.sub(r'\*\*(.*?)\*\*', r'\1', text)
        text = re.sub(r'\*(.*?)\*', r'\1', text)
        
        # Replace markdown bullets with clean bullets
        text = re.sub(r'^\s*[-*]\s+', '• ', text, flags=re.MULTILINE)
        
        # Remove any other markdown symbols
        text = re.sub(r'#+\s*', '', text)  # Headers
        text = re.sub(r'`(.*?)`', r'\1', text)  # Code
        text = re.sub(r'\[(.*?)\]\(.*?\)', r'\1', text)  # Links
        
        # Ensure proper spacing between paragraphs
        text = re.sub(r'\n\s*\n', '\n\n', text)
        
        return text.strip()

    def _build_intelligent_prompt(self, context: str, question: str, format_type: str) -> str:
        """Build dynamic prompt based on analyzed format needs"""
        
        base_prompt = """You are an expert consultant and strategic advisor. Provide intelligent, nuanced responses that demonstrate true understanding.

        else:  # bullet format

• Clear, concise bullet points
• Each bullet can have 1-2 sentences of explanation
• Group related items together

"""

        # CONTENT ENHANCEMENT FOR DIFFERENT QUESTION TYPES
        if 'approach' in question.lower() or 'framework' in question.lower():
            base_prompt += """

FOR APPROACH/FRAMEWORK QUESTIONS:
Provide a structured yet concise response that covers:

Overall Methodology: Briefly explain the core approach and philosophy.

Key Phases/Components: Use bullet points for clear structure of main elements.

    def generate_answer(self, context: str, question: str, conversation_history: str = "") -> str:
        if not self.loaded:
            return "Gemini is not available right now."

        try:
            # STEP 1: Intelligently analyze the question format needs
            format_type = self._analyze_question_intent(question)
            print(f"🎯 Detected format type: {format_type}")
            
            # STEP 2: Build dynamic prompt based on format needs
            if conversation_history:
               prompt = f"""Previous conversation context:
            {conversation_history}

            Current question: {question}

            Relevant document context:
            {context if context and "No relevant information" not in context else "No specific document context available"}

            Please answer the current question while maintaining continuity with the previous conversation:"""
            else:
               prompt = self._build_intelligent_prompt(context, question, format_type)
            
            # STEP 3: Generate response
            response = self.model.generate_content(prompt)
            
            # STEP 4: Clean the response formatting
            answer = self._clean_response_formatting(response.text.strip())
            
            # STEP 5: Ensure reasonable length and quality
            if not answer or len(answer) < 10:
                return "I apologize, but I couldn't generate a meaningful response. Could you please rephrase your question?"
            
            # If response is too long, try to shorten it
            if len(answer) > 1500:
                print("⚠️ Response too long, applying smart shortening...")
                # Keep only the most relevant parts
                sentences = answer.split('. ')
                if len(sentences) > 15:
                    answer = '. '.join(sentences[:12]) + '.'
            
            return answer

        except Exception as e:
            print(f"Gemini API error: {e}")
            return "I encountered an issue while processing your request. Please try again."

def get_file_hash(file_path: str) -> str:
    hasher = hashlib.md5()
    with open(file_path, 'rb') as f:
        hasher.update(f.read())
    return hasher.hexdigest()

def get_folder_hash(folder_url: str) -> str:
    return hashlib.md5(folder_url.encode()).hexdigest()

def chunk_text(text: str, chunk_size: int = 1000, overlap: int = 100) -> List[str]:
    """Improved chunking with overlap for better context."""
    sentences = re.split(r'(?<=[.!?])\s+', text)
    chunks = []
    current_chunk = ""

    for sentence in sentences:
        if len(current_chunk) + len(sentence) + 1 > chunk_size and current_chunk:
            chunks.append(current_chunk.strip())
            # Add overlap: keep last part of current chunk
            overlap_start = max(0, len(current_chunk) - overlap)
            current_chunk = current_chunk[overlap_start:] + " " + sentence
        else:
            current_chunk += " " + sentence if current_chunk else sentence

    if current_chunk:
        chunks.append(current_chunk.strip())

    return chunks

def extract_folder_id_from_url(url: str) -> str:
    patterns = [
        r'https://drive\.google\.com/drive/folders/([a-zA-Z0-9_-]+)',
        r'https://docs\.google\.com/document/d/([a-zA-Z0-9_-]+)',
        r'folders/([a-zA-Z0-9_-]+)'
    ]
    for pattern in patterns:
        match = re.search(pattern, url)
        if match:
            return match.group(1)
    raise ValueError("Invalid Google Drive folder URL")

def get_drive_service(request: Request):
    """Get Drive service with COMPLETE token handling - FIXED VERSION"""
    user = request.session.get('user')
    if not user or 'access_token' not in user:
        raise HTTPException(status_code=401, detail="Not authenticated")

    try:
        # Load client secrets to get ALL required fields
        with open(CLIENT_SECRETS_FILE, 'r') as f:
            client_secrets = json.load(f)
        
        client_id = client_secrets['web']['client_id']
        client_secret = client_secrets['web']['client_secret'] 
        token_uri = client_secrets['web']['token_uri']
        
        print(f"✅ Loaded client config: {client_id[:20]}...")

    except Exception as e:
        print(f"❌ Error loading client secrets: {e}")
        raise HTTPException(status_code=500, detail="OAuth configuration error")

    # Get refresh token from database
    conn = get_db_connection()
    cursor = conn.cursor()
    cursor.execute("SELECT refresh_token FROM users WHERE email = ?", (user['email'],))
    row = cursor.fetchone()
    refresh_token = row[0] if row else None
    conn.close()

    if not refresh_token:
        print("❌ No refresh token found in database")
        # Force re-authentication
        request.session.clear()
        raise HTTPException(status_code=401, detail="Session expired - please login again")

    print(f"🔄 Creating credentials with refresh_token: {refresh_token[:20]}...")

    # Create credentials with ALL required fields for token refresh
    creds = Credentials(
        token=user['access_token'],
        refresh_token=refresh_token,
        token_uri=token_uri,
        client_id=client_id,
        client_secret=client_secret,
        scopes=SCOPES
    )
    
    print(f"🔍 Token valid: {creds.valid}, expired: {creds.expired}")
    
    # ALWAYS refresh token to ensure it's fresh (preventative)
    try:
        print("🔄 Proactively refreshing access token...")
        creds.refresh(GoogleAuthRequest())
        print("✅ Token refreshed successfully")
        
        # Update session with new token
        user['access_token'] = creds.token
        request.session['user'] = user
        
        # Update database with new token
        conn = get_db_connection()
        cursor = conn.cursor()
        cursor.execute("UPDATE users SET access_token = ? WHERE email = ?", 
                     (creds.token, user['email']))
        conn.commit()
        conn.close()
        print("✅ Database updated with fresh access token")
        
    except Exception as refresh_error:
        print(f"❌ Token refresh failed: {refresh_error}")
        # Clear session to force re-login
        request.session.clear()
        raise HTTPException(status_code=401, detail="Authentication expired - please login again")

    return build("drive", "v3", credentials=creds)

def get_files_from_drive_folder(drive_service, folder_id: str) -> List[Dict]:
    try:
        query = f"'{folder_id}' in parents and trashed=false"
        results = drive_service.files().list(
            q=query,
            pageSize=200,
            fields="files(id, name, size, mimeType, modifiedTime)"
        ).execute()
        files = []
        for f in results.get('files', []):
            # Add support for PowerPoint files
            supported_mime_types = [
                'application/pdf', 
                'application/vnd.google-apps.document', 
                'text/plain',
                'application/vnd.ms-powerpoint',
                'application/vnd.openxmlformats-officedocument.presentationml.presentation'
            ]
            supported_extensions = ['.pdf', '.docx', '.doc', '.txt', '.ppt', '.pptx']
            
            if (f['mimeType'] in supported_mime_types or
                any(f['name'].lower().endswith(ext) for ext in supported_extensions)):
                files.append({
                    'id': f['id'],
                    'name': f['name'],
                    'size': int(f.get('size', 0)),
                    'mimeType': f['mimeType'],
                    'modifiedTime': f.get('modifiedTime', '')
                })
        return files
    except Exception as e:
        print(f"Error fetching files from Drive: {e}")
        raise

def download_drive_file(drive_service, file_id: str, file_name: str, mime_type: str) -> str:
    try:
        safe_filename = "".join(c for c in file_name if c.isalnum() or c in (' ', '-', '_', '.')).rstrip()
        base_file_path = f"uploaded_files/{file_id}_{safe_filename}"
        is_pdf_export = False

        if mime_type == 'application/vnd.google-apps.document':
            request_media = drive_service.files().export_media(fileId=file_id, mimeType='application/pdf')
            file_path = base_file_path.split('.')[0] + '.pdf'
            is_pdf_export = True
        else:
            request_media = drive_service.files().get_media(fileId=file_id)
            file_path = base_file_path

        if is_pdf_export and not file_path.lower().endswith('.pdf'):
            file_path += '.pdf'
        elif not is_pdf_export and '.' not in file_path:
            if 'text' in mime_type:
                file_path += '.txt'
            elif 'pdf' in mime_type:
                file_path += '.pdf'
            elif 'powerpoint' in mime_type or 'presentation' in mime_type:
                file_path += '.pptx'

        with open(file_path, 'wb') as fh:
            downloader = MediaIoBaseDownload(fh, request_media)
            done = False
            while not done:
                status, done = downloader.next_chunk()

        return file_path
    except Exception as e:
        print(f"Error downloading file {file_name}: {e}")
        raise

def process_file(file_path: str, file_name: str, user_id: int, folder_id: int, drive_file_id: str) -> bool:
    try:
        text = ""
        try:
            if file_path.endswith('.txt'):
                with open(file_path, 'r', encoding='utf-8') as f:
                    text = f.read()
            elif file_path.endswith('.pdf'):
                with open(file_path, 'rb') as f:
                    reader = PyPDF2.PdfReader(f)
                    for page in reader.pages:
                        page_text = page.extract_text()
                        if page_text:
                            text += page_text + '\n'
            elif file_path.endswith(('.doc', '.docx')):
                doc = Document(file_path)
                text = '\n'.join([p.text for p in doc.paragraphs])
            elif file_path.endswith(('.ppt', '.pptx')):
                # Process PowerPoint files
                presentation = Presentation(file_path)
                for slide in presentation.slides:
                    for shape in slide.shapes:
                        if hasattr(shape, "text"):
                            text += shape.text + '\n'
            else:
                print(f"⚠️ Unsupported file type: {file_name}")
                return False
        except Exception as e:
            print(f"File reading/parsing error for {file_name}: {e}")
            return False

        if not text.strip():
            print(f"⚠️ No text extracted from {file_name}")
            return False

        chunks = chunk_text(text)
        if not chunks:
            print(f"⚠️ No chunks created from {file_name}")
            return False

        print(f"📄 Processing {file_name}: {len(chunks)} chunks, {len(text)} characters")

        # Generate embeddings with proper verification
        embeddings = embedding_model.encode(chunks)
        print(f"📐 Generated {len(embeddings)} embeddings, dimension: {len(embeddings[0])}")

        # Verify embedding dimension matches Pinecone index
        if len(embeddings[0]) != DIMENSION:
            print(f"❌ Embedding dimension mismatch: Expected {DIMENSION}, got {len(embeddings[0])}")
            return False

        conn = get_db_connection()
        cursor = conn.cursor()
        file_hash = get_file_hash(file_path)

        # FIXED: Use INSERT OR IGNORE to prevent DB errors without skipping processing
        cursor.execute(

# -------------------------------------------------------------------
def semantic_search(question: str, n_results: int = 5):
    try:
        print(f"🔍 Searching Pinecone for: {question}")
        print(f"🔍 Encoding query: '{question}'")
        query_embedding = embedding_model.encode([question]).tolist()[0]
        print(f"📐 Embedding dimension: {len(query_embedding)}")
        
        # Query Pinecone with proper parameters
        results = index.query(
            vector=query_embedding, 
            top_k=n_results, 
            include_metadata=True,
            include_values=False
        )
        
        print(f"📊 Raw Pinecone results: {len(results.matches)} matches")
        
        documents = []
        scores = []
        
        for i, match in enumerate(results.matches):
            score = match.score
            metadata = match.metadata or {}
            
            print(f"   Match {i+1}: Score={score:.3f}, File={metadata.get('file_name', 'Unknown')}")
            
            # Extract text from metadata
            chunk_text = metadata.get("chunk_text") or metadata.get("text") or ""
            
            # Use a reasonable similarity threshold (increased from 0.3)
            if chunk_text and score > 0.35:  # Increased threshold for better quality
                documents.append({
                    "text": chunk_text,
                    "score": score,
                    "file_name": metadata.get("file_name", "Unknown"),
                    "chunk_index": metadata.get("chunk_index", 0)
                })
                scores.append(score)
                print(f"   ✅ Added document {i+1} (score: {score:.3f})")
            else:
                print(f"   ❌ Skipped document {i+1} (score: {score:.3f})")
        
        print(f"📚 Final relevant documents: {len(documents)}")
        return {"documents": documents, "scores": scores}
        
    except Exception as e:
        print(f"❌ Pinecone search error: {e}")
        print(f"🔍 Full traceback: {traceback.format_exc()}")
        return {"documents": [], "scores": []}

def process_file_in_background(file_info: Dict[str, Any], user_id: int, folder_id: int, drive_service) -> Dict[str, Any]:
    file_name = file_info['name']
    try:
        print(f"📥 Downloading {file_name}...")
        file_path = download_drive_file(drive_service, file_info['id'], file_info['name'], file_info['mimeType'])
        print(f"✅ Downloaded {file_name} to {file_path}")
        
        if process_file(file_path, file_name, user_id, folder_id, file_info['id']):
            return {"status": "success", "file_name": file_name}
        else:
            return {"status": "error", "file_name": file_name, "message": "Failed to process (empty/unsupported)"}
    except Exception as e:
        return {"status": "error", "file_name": file_name, "message": str(e)}

def process_all_selected_files_sync(folder_data, selected_file_ids, user_id, folder_id, drive_service) -> Tuple[int, List[str]]:
    processed_count = 0
    errors = []
    for file_info in folder_data['files']:
        if file_info['id'] in selected_file_ids:
            print(f"🔄 Processing {file_info['name']}...")
            res = process_file_in_background(file_info, user_id, folder_id, drive_service)
            if res['status'] == 'success':
                processed_count += 1
                print(f"✅ Successfully processed {res['file_name']}")
            else:
                errors.append(f"Error processing {res['file_name']}: {res.get('message')}")
                print(f"❌ Failed to process {res['file_name']}: {res.get('message')}")
    return processed_count, errors

        }
    except Exception as e:
        return {"error": str(e)}

@app.get("/debug-file-processing")
async def debug_file_processing():
    """Debug endpoint to check file processing status"""
    try:
        conn = get_db_connection()
        cursor = conn.cursor()
        
        # Check files in database
        cursor.execute('SELECT COUNT(*) as file_count, SUM(file_size) as total_size FROM files')
        file_stats = cursor.fetchone()
        
        cursor.execute('''
            SELECT f.file_name, f.file_size, f.processed_at, COUNT(c.chunk_id) as chunk_count 
            FROM files f 
            LEFT JOIN chunks c ON f.file_id = c.file_id 
            GROUP BY f.file_id 
            LIMIT 10
        ''')
        files_with_chunks = cursor.fetchall()
        
        # Check Pinecone stats
        pinecone_stats = index.describe_index_stats()
        
        conn.close()
        
        return {
            "database_files": {
                "total_files": file_stats[0] or 0,
                "total_size_bytes": file_stats[1] or 0,
                "sample_files": [
                    {
                        "name": row[0],
                        "size": row[1],
                        "processed_at": row[2],
                        "chunks": row[3]
                    } for row in files_with_chunks
                ]
            },
            "pinecone_vectors": pinecone_stats.total_vector_count,
            "uploaded_files_directory": {
                "exists": Path("uploaded_files").exists(),
                "file_count": len(list(Path("uploaded_files").glob("*"))) if Path("uploaded_files").exists() else 0
            }
        }
        
    except Exception as e:
        return {"error": str(e)}
